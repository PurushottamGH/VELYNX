python# SQLite-backed domain concept store
# Separate DB: data/knowledge_graph.db (never touch data/graph.db)
#
# Schema:
# TABLE concepts: id INTEGER PK, name TEXT UNIQUE, domain TEXT, description TEXT
# TABLE relationships: id INTEGER PK, source TEXT, relation TEXT, target TEXT
#   relation types: "requires", "produces", "opposes", "enables", "part_of"
#
# Seed with these domain concepts on first build:
#
# CODING domain:
#   debugging       — isolating why code behaves unexpectedly
#   refactoring     — restructuring code without changing behavior
#   testing         — verifying behavior matches intention
#   architecture    — structuring systems for change and scale
#   iteration       — repeated cycles of build-test-improve
#   deployment      — releasing code to a live environment
#   optimization    — improving performance within constraints
#   abstraction     — hiding complexity behind a clean interface
#   technical debt  — accumulated cost of shortcuts taken earlier
#   legacy code     — existing code that works but is hard to change
#
# LOGIC domain:
#   proof           — establishing truth through reasoning
#   inference       — deriving conclusions from evidence
#   contradiction   — two claims that cannot both be true
#   assumption      — a belief taken as true without verification
#   verification    — confirming a claim against evidence
#
# SYSTEMS domain:
#   bottleneck      — the constraint limiting total throughput
#   failure         — a system not meeting its intended behavior
#   recovery        — restoring function after failure
#   fallback        — a backup path when primary fails
#   feedback loop   — output influencing its own input

    "trust":       ["verification", "testing", "proof"],
    "pain":        ["technical debt", "bottleneck", "debugging"],
    "identity":    ["architecture", "abstraction"],
    "courage":     ["deployment", "refactoring"],
    "grief":       ["legacy code", "technical debt"],
    "anger":       ["bottleneck", "technical debt"],
    "hope":        ["recovery", "feedback loop"],
    "understanding": ["inference", "abstraction", "proof"],
    "regret":      ["technical debt", "assumption"],
    "forgiveness": ["refactoring", "recovery"],
}

def get_domain_concepts(soul_concept: str) -> list[str]:
    """Return domain concepts bridged from a soul concept."""
    return BRIDGE_NODES.get(soul_concept, [])

def get_soul_concepts(domain_concept: str) -> list[str]:
    """Reverse bridge — which soul concepts connect to this domain concept."""
    return [s for s, targets in BRIDGE_NODES.items() if domain_concept in targets]

def bridge_exists(soul_concept: str, domain_concept: str) -> bool:
    return domain_concept in BRIDGE_NODES.get(soul_concept, [])
